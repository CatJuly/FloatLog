```python
from __future__ import annotations

import os
from dataclasses import dataclass, fields, asdict
from pathlib import Path


@dataclass
class Server:
    name: str
    host: str
    username: str
    auth_type: str = "password"  # "password" | "key"
    secret: str = ""
    port: int = 22
    deploy_dir: str = ""
    log_dir: str = ""
    note: str = ""

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: dict) -> "Server":
        names = {f.name for f in fields(cls)}
        return cls(**{k: v for k, v in data.items() if k in names})


@dataclass
class AppConfig:
    workspace_dir: str
    claude_command: str = "claude"
    permission_mode: str = "bypassPermissions"
    timeout_seconds: int = 600

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: dict) -> "AppConfig":
        defaults = default_config()
        merged = {**defaults.to_dict(), **data}
        names = {f.name for f in fields(cls)}
        return cls(**{k: merged[k] for k in names})


def default_data_dir() -> Path:
    base = os.environ.get("APPDATA") or str(Path.home())
    return Path(base) / "FloatLog"


def default_workspace_dir() -> Path:
    return default_data_dir() / "workspace"


def default_config() -> AppConfig:
    return AppConfig(workspace_dir=str(default_workspace_dir()))
```

from __future__ import annotations

import json
from pathlib import Path

from models import Server, AppConfig, default_config, default_data_dir


class Storage:
    def __init__(self, data_dir: Path | None = None) -> None:
        self.data_dir = Path(data_dir) if data_dir else default_data_dir()
        self.servers_path = self.data_dir / "servers.json"
        self.config_path = self.data_dir / "config.json"

    def ensure_dirs(self) -> None:
        self.data_dir.mkdir(parents=True, exist_ok=True)

    def list_servers(self) -> list[Server]:
        if not self.servers_path.exists():
            return []
        raw = json.loads(self.servers_path.read_text(encoding="utf-8"))
        return [Server.from_dict(item) for item in raw]

    def save_servers(self, servers: list[Server]) -> None:
        self.ensure_dirs()
        data = [s.to_dict() for s in servers]
        self.servers_path.write_text(
            json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8"
        )

    def load_config(self) -> AppConfig:
        if not self.config_path.exists():
            return default_config()
        raw = json.loads(self.config_path.read_text(encoding="utf-8"))
        return AppConfig.from_dict(raw)

    def save_config(self, config: AppConfig) -> None:
        self.ensure_dirs()
        self.config_path.write_text(
            json.dumps(config.to_dict(), ensure_ascii=False, indent=2), encoding="utf-8"
        )
```

from __future__ import annotations

from dataclasses import dataclass

from models import Server


@dataclass
class ClaudeResult:
    status: str  # "success" | "not_found" | "timeout" | "error"
    output: str
    error: str = ""

    @property
    def ok(self) -> bool:
        return self.status == "success"


def _auth_desc(server: Server) -> str:
    if server.auth_type == "key":
        return f"私钥路径: {server.secret}"
    return f"密码: {server.secret}"


def build_prompt(servers: list[Server], user_prompt: str) -> str:
    parts = ["请连接下面这些服务器，查看错误日志并分析：", ""]
    for i, server in enumerate(servers, start=1):
        parts.append(f"服务器 {i}：")
        parts.append(f"- 名称: {server.name}")
        parts.append(f"- 主机: {server.host}:{server.port}")
        parts.append(f"- 用户: {server.username}")
        parts.append(f"- 认证: {_auth_desc(server)}")
        if server.deploy_dir:
            parts.append(f"- 部署位置: {server.deploy_dir}")
        if server.log_dir:
            parts.append(f"- 日志目录: {server.log_dir}")
        parts.append("")
    parts.append(f"用户需求: {user_prompt}")
    return "\n".join(parts).rstrip() + "\n"
```

def resolve_binary(command: str) -> str | None:
    return shutil.which(command)


def _resolve_windows_shim(resolved: str) -> str:
    if os.path.splitext(resolved)[1]:
        return resolved
    for ext in (".exe", ".com", ".cmd", ".bat"):
        sibling = f"{resolved}{ext}"
        if os.path.isfile(sibling):
            return sibling
    return resolved


def build_command(config: AppConfig) -> list[str]:
    return [
        config.claude_command,
        "--print",
        "--output-format", "text",
        "--input-format", "text",
        "--add-dir", config.workspace_dir,
        "--permission-mode", config.permission_mode,
    ]


def launch_command(cmd: list[str]) -> list[str]:
    executable = cmd[0]
    resolved = shutil.which(executable)
    if not resolved:
        return cmd
    if os.name == "nt":
        resolved = _resolve_windows_shim(resolved)
        if os.path.splitext(resolved)[1].lower() in {".cmd", ".bat"}:
            comspec = os.environ.get("COMSPEC") or shutil.which("cmd.exe") or "cmd.exe"
            return [comspec, "/d", "/s", "/c", resolved, *cmd[1:]]
    if resolved == executable:
        return cmd
    return [resolved, *cmd[1:]]
```

def run(config: AppConfig, servers: list[Server], user_prompt: str) -> ClaudeResult:
    if resolve_binary(config.claude_command) is None:
        return ClaudeResult(
            status="not_found",
            output="",
            error=f"未找到命令: {config.claude_command}",
        )

    prompt = build_prompt(servers, user_prompt)
    cmd = build_command(config)
    launch = launch_command(cmd)

    workspace = Path(config.workspace_dir)
    workspace.mkdir(parents=True, exist_ok=True)

    try:
        proc = subprocess.run(
            launch,
            input=prompt,
            capture_output=True,
            text=True,
            encoding="utf-8",
            cwd=str(workspace),
            timeout=config.timeout_seconds,
        )
    except subprocess.TimeoutExpired:
        return ClaudeResult(
            status="timeout",
            output="",
            error=f"运行超时（{config.timeout_seconds} 秒）",
        )
    except OSError as exc:
        return ClaudeResult(status="error", output="", error=str(exc))

    output = proc.stdout or ""
    err = proc.stderr or ""
    if proc.returncode == 0:
        return ClaudeResult(status="success", output=output, error=err)
    return ClaudeResult(
        status="error",
        output=output,
        error=err or f"claude 退出码 {proc.returncode}",
    )
```

from PySide6.QtCore import Qt, QPoint, QThread, Signal
from PySide6.QtWidgets import (

class TitleBar(QWidget):
    def __init__(self, parent=None) -> None:
        super().__init__(parent)
        self._drag_pos: QPoint | None = None
        layout = QHBoxLayout(self)
        layout.setContentsMargins(8, 4, 8, 4)

        title = QLabel("FloatLog")
        title.setStyleSheet("font-weight: bold;")

        self.pin_check = QCheckBox("置顶")
        self.pin_check.toggled.connect(self._toggle_pin)

        self.min_btn = QPushButton("—")
        self.min_btn.setFixedWidth(24)
        self.min_btn.clicked.connect(lambda: self.window().showMinimized())

        self.close_btn = QPushButton("×")
        self.close_btn.setFixedWidth(24)
        self.close_btn.clicked.connect(self._close_to_tray)

        layout.addWidget(title)
        layout.addStretch(1)
        layout.addWidget(self.pin_check)
        layout.addWidget(self.min_btn)
        layout.addWidget(self.close_btn)

    def _toggle_pin(self, on: bool) -> None:
        self.window().setWindowFlag(Qt.WindowStaysOnTopHint, on)
        self.window().show()

    def _close_to_tray(self) -> None:
        self.window().hide()

    def mousePressEvent(self, event) -> None:
        if event.button() == Qt.LeftButton:
            self._drag_pos = event.globalPosition().toPoint() - self.window().frameGeometry().topLeft()
            event.accept()

    def mouseMoveEvent(self, event) -> None:
        if self._drag_pos is not None and event.buttons() & Qt.LeftButton:
            self.window().move(event.globalPosition().toPoint() - self._drag_pos)
            event.accept()

    def mouseReleaseEvent(self, event) -> None:
        self._drag_pos = None


class MainWindow(QWidget):
    def __init__(self, storage: Storage, config: AppConfig) -> None:
        super().__init__()
        self.storage = storage
        self.config = config
        self.setWindowFlags(Qt.FramelessWindowHint | Qt.Tool)
        self.resize(380, 540)
        self.setWindowTitle("FloatLog")

        layout = QVBoxLayout(self)
        layout.setContentsMargins(0, 0, 0, 0)
        layout.setSpacing(0)
        layout.addWidget(TitleBar(self))

        body = QVBoxLayout()
        body.setContentsMargins(8, 8, 8, 8)

        self.server_list = QListWidget()
        self.server_list.setSelectionMode(QAbstractItemView.ExtendedSelection)
        body.addWidget(self.server_list, stretch=3)

        list_btns = QHBoxLayout()

        add_btn.clicked.connect(self.add_server)

        del_btn = QPushButton("删除")
        del_btn.clicked.connect(self.delete_server)
        settings_btn = QPushButton("设置")
        settings_btn.clicked.connect(self.open_settings)
        list_btns.addWidget(add_btn)
        list_btns.addWidget(edit_btn)
        list_btns.addWidget(del_btn)
        list_btns.addWidget(settings_btn)
        body.addLayout(list_btns)

        body.addWidget(QLabel("提示词（要 Claude Code 做什么）："))
        self.prompt_edit = QPlainTextEdit()
        self.prompt_edit.setFixedHeight(80)
        self.prompt_edit.setPlaceholderText("例如：分析今天的错误日志")
        body.addWidget(self.prompt_edit)

        self.send_btn = QPushButton("发送分析")
        self.send_btn.clicked.connect(self.send)
        body.addWidget(self.send_btn)

        body.addWidget(QLabel("结果："))
        self.result_view = QPlainTextEdit()
        self.result_view.setReadOnly(True)
        self.result_view.setPlaceholderText("分析结果将显示在这里")
        body.addWidget(self.result_view, stretch=4)

        copy_btn = QPushButton("复制结果")
        copy_btn.clicked.connect(self.copy_result)
        body.addWidget(copy_btn)

        layout.addLayout(body)

        self.refresh_list()

    # ── 服务器列表 ──────────────────────────────
    def refresh_list(self) -> None:
        self.server_list.clear()
        for server in self.storage.list_servers():
            item = QListWidgetItem(f"{server.name}  ({server.host}:{server.port})")
            item.setData(Qt.UserRole, server)
            self.server_list.addItem(item)

    def _selected_servers(self) -> list[Server]:
        return [it.data(Qt.UserRole) for it in self.server_list.selectedItems()]

    def add_server(self) -> None:
        dialog = ServerDialog(self)
        if dialog.exec():
            servers = self.storage.list_servers()
            servers.append(dialog.result_server())
            self.storage.save_servers(servers)
            self.refresh_list()

    def edit_server(self) -> None:
        items = self.server_list.selectedItems()
        if not items:
            QMessageBox.information(self, "提示", "请先选择一台服务器")
            return
        current = items[0].data(Qt.UserRole)
        dialog = ServerDialog(self, current)
        if dialog.exec():
            new = dialog.result_server()
            servers = self.storage.list_servers()
            for i, s in enumerate(servers):
                if s == current:
                    servers[i] = new
                    break
            self.storage.save_servers(servers)
            self.refresh_list()

    def delete_server(self) -> None:
        selected = self._selected_servers()
        if not selected:
            QMessageBox.information(self, "提示", "请先选择要删除的服务器")
            return
        if QMessageBox.question(self, "确认",
                                f"删除选中的 {len(selected)} 台服务器？") != QMessageBox.Yes:
            return
        servers = [s for s in self.storage.list_servers() if s not in selected]
        self.storage.save_servers(servers)
        self.refresh_list()

    def open_settings(self) -> None:
        dialog = SettingsDialog(self, self.config)
        if dialog.exec():
            self.config = dialog.result_config()
            self.storage.save_config(self.config)

    # ── 发送分析 ─────────────────────────────────
    def send(self) -> None:
        servers = self._selected_servers()
        if not servers:
            QMessageBox.warning(self, "提示", "请先选择至少一台服务器")
            return
        user_prompt = self.prompt_edit.toPlainText().strip()
        if not user_prompt:
            QMessageBox.warning(self, "提示", "请输入提示词")
            return

        self.send_btn.setEnabled(False)
        self.send_btn.setText("运行中…")
        self.result_view.setPlainText("正在分析，请稍候…")

        if getattr(self, "_worker", None) is not None and self._worker.isRunning():
            self._worker.wait()

        self._worker = Worker(self.config, servers, user_prompt)
        self._worker.finished.connect(self._on_finished)
        self._worker.start()

    def _on_finished(self, result) -> None:
        self.send_btn.setEnabled(True)
        self.send_btn.setText("发送分析")
        if result.ok:
            self.result_view.setPlainText(result.output or "(无输出)")
        else:
            self.result_view.setPlainText(f"[{result.status}] {result.error}")

    def copy_result(self) -> None:
        from PySide6.QtWidgets import QApplication
        QApplication.clipboard().setText(self.result_view.toPlainText())


class Worker(QThread):
    finished = Signal(object)

    def __init__(self, config: AppConfig, servers: list[Server], user_prompt: str) -> None:
        super().__init__()
        self._config = config
        self._servers = servers
        self._user_prompt = user_prompt

    def run(self) -> None:
        result = run_claude(self._config, self._servers, self._user_prompt)
        self.finished.emit(result)

from storage import Storage

def make_icon() -> QIcon:
    pm = QPixmap(64, 64)
    pm.fill(Qt.transparent)
    painter = QPainter(pm)
    painter.setRenderHint(QPainter.Antialiasing)
    painter.setBrush(QColor("#3b82f6"))
    painter.setPen(Qt.NoPen)
    painter.drawRoundedRect(6, 6, 52, 52, 12, 12)
    painter.setPen(QColor("white"))
    font = painter.font()
    font.setBold(True)
    font.setPixelSize(28)
    painter.setFont(font)
    painter.drawText(pm.rect(), Qt.AlignCenter, "L")
    painter.end()
    return QIcon(pm)


def main() -> int:
    app = QApplication(sys.argv)
    app.setQuitOnLastWindowClosed(False)

    storage = Storage()
    storage.ensure_dirs()
    config = storage.load_config()

    window = MainWindow(storage, config)

    tray = QSystemTrayIcon(make_icon(), app)
    menu = QMenu()
    show_action = QAction("显示")
    show_action.triggered.connect(window.show)
    quit_action = QAction("退出")
    quit_action.triggered.connect(app.quit)
    menu.addAction(show_action)
    menu.addAction(quit_action)
    tray.setContextMenu(menu)
    tray.activated.connect(
        lambda reason: window.show() if reason == QSystemTrayIcon.Trigger else None
    )
    tray.show()

    window.show()
    return app.exec()


if __name__ == "__main__":
    sys.exit(main())
